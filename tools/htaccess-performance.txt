# Reglages serveur a ajouter dans le .htaccess du SERVEUR, juste apres le bloc "# END Securite".
#
# Ce fichier n'est pas dans Git (voir .gitignore) : il faut l'editer a la main sur le serveur. Ne PAS enregistrer
# de fiche dans le back-office sans en avoir fait une copie avant : le generateur seo() reecrit le fichier en entier.
# Chaque bloc est protege par <IfModule> : sans effet si le module n'existe pas.

# 1. Cache long pour les types qui n'avaient pas de regle (webp = 30 jours par defaut aujourd'hui)
<IfModule mod_expires.c>
  ExpiresByType image/webp "access plus 1 year"
  ExpiresByType image/avif "access plus 1 year"
  ExpiresByType image/svg+xml "access plus 1 year"
  ExpiresByType font/woff2 "access plus 1 year"
  ExpiresByType font/woff "access plus 1 year"
  ExpiresByType application/font-woff2 "access plus 1 year"
  ExpiresByType video/mp4 "access plus 1 year"
</IfModule>

# 2. Compression des types texte qui n'y etaient pas (SVG, JS moderne, JSON, polices TTF/EOT)
<IfModule mod_deflate.c>
  AddOutputFilterByType DEFLATE image/svg+xml text/javascript application/javascript application/json application/ld+json font/ttf application/vnd.ms-fontobject
</IfModule>
