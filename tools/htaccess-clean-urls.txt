# =====================================================================================
# URLs propres : /index.php et /index.php?option=... ne doivent JAMAIS etre visibles.
# A coller dans le .htaccess RACINE DU SERVEUR (hors Git), juste apres "RewriteEngine on"
# (et apres la redirection https/www), AVANT toutes les autres RewriteRule.
# Sauvegardez d'abord :  cp .htaccess .htaccess.bak-urls   (et supprimez la copie apres test)
# Le test sur THE_REQUEST ne regarde que la requete ENVOYEE PAR LE NAVIGATEUR : les reecritures
# internes (/agents-ia/ -> index.php?option=...) ne sont pas touchees, pas de boucle.
# =====================================================================================

# 1) /index.php (sans parametres) -> /

RewriteCond %{REQUEST_METHOD} ^(GET|HEAD)$
RewriteRule ^index\.php$ / [R=301,L]
